Script Python - Chat DeepSeek

import opendeep
from opendeep import GenerativeModel
import re

TOKEN_DEEPSEEK = "SEU_USER_TOKENS"

# Configura a conexão e o modelo
opendeep.configure(api_key=TOKEN_DEEPSEEK)
model = GenerativeModel("deepseek-v4-pro")

print("=" * 45)
print("🤖 Chat com DeepSeek iniciado! (Digite 'sair' para encerrar)")
print("=" * 45)

while True:
    pergunta = input("\nVocê: ")
    
    # Condição para fechar o chat
    if pergunta.strip().lower() in ["sair", "exit", "quit"]:
        print("Encerrando o chat...")
        break
    
    if not pergunta.strip():
        continue
    
    print("DeepSeek pensando...")
    
    # Envia a mensagem para a IA
    resposta = model.generate_content(pergunta)
    
    # Extrai o texto da resposta
    texto = resposta.text if hasattr(resposta, "text") else str(resposta)
    
    # Remove as tags internas de pensamento (...) caso a IA mande
    texto_limpo = re.sub(r'<think>.*?</think>', '', texto, flags=re.DOTALL).strip()
    
    print(f"\nDeepSeek: {texto_limpo}")
